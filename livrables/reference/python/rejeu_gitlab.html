rejeu_gitlab

Outil historique de reconstruction d'un plan MiniShop en 32 commits commentés.

⚠️ Ce plan décrit un état antérieur (notamment l'ancienne CI GitLab) ; il ne constitue ni un journal Git constaté ni l'état courant du dépôt. La fiche générée avec --fiche reste une trace de ce plan ancien. Le mode --dest crée une histoire synthétique dans une copie distincte : ne pas l'exécuter pour fabriquer ou pousser un historique sans autorisation explicite.

Deux usages :
  1. Générer la fiche de suivi (documentation humaine) : python3 scripts/rejeu_gitlab.py --fiche docs/08-fiche-suivi-commits.md
  2. Reconstruire un dépôt neuf, commit par commit, dans une COPIE isolée (le dépôt courant n'est jamais modifié) : python3 scripts/rejeu_gitlab.py --dest ../minishop-sae --nom "Prénom Nom" --email prenom.nom@etu.univ-cotedazur.fr # puis, une fois le projet vide créé sur GitLab : cd ../minishop-sae git remote add origin https://gitlab.example.fr//minishop-sae.git git push -u origin main --tags

Le plan (ci-dessous) est la source unique de vérité : la fiche est générée depuis ce même plan, les deux ne peuvent pas diverger.

458def message_complet(etape: dict) -> str:
459    r"""Assemble le message de commit complet d'une étape de rejeu.

461    Args:
462        etape: Dictionnaire contenant les clés ``titre`` et ``corps``.
463
464    Returns:
465        Chaîne multi-lignes prête pour ``git commit``.
466

468        >>> message_complet({"titre": "feat: init", "corps": ["Base prête."]})
469        'feat: init\n\nBase prête.'
470    """
471    return "\n\n".join([etape["titre"], *etape["corps"]])

636def git(cwd: str, *args: str) -> subprocess.CompletedProcess:
637    """Exécute une sous-commande Git dans un dépôt cible et échoue si le retour diffère de 0.
638
639    Args:
640        cwd: Chemin du dépôt Git cible.
641        *args: Arguments transmis à ``git``.
642
643    Returns:
644        Résultat ``CompletedProcess`` de l'exécution Git.
645
646    Raises:
647        SystemExit: Si la commande Git retourne un code non nul.
648    """
649    r = subprocess.run(["git", "-C", cwd, *args], capture_output=True, text=True)
650    if r.returncode != 0:
651        raise SystemExit(f"git {' '.join(args)} a échoué dans {cwd} :\n{r.stderr}")
652    return r

655def rejouer(dest: str, nom: str, email: str) -> None:
656    """Extrait HEAD dans un dossier vierge et rejoue l'historique des 32 commits.
657
658    Args:
659        dest: Dossier cible vierge où créer le dépôt Git reconstruit.
660        nom: Nom d'auteur utilisé pour les commits rejoués.
661        email: Adresse courriel utilisée pour les commits rejoués.
662
663    Raises:
664        SystemExit: Si ``dest`` est non vide ou si la couverture finale diffère de ``HEAD``.
665    """
666    dest = os.path.abspath(dest)
667    if os.path.exists(dest) and os.listdir(dest):
668        raise SystemExit(f"Le dossier {dest} existe et n'est pas vide — choisir un --dest vierge.")
669
670    os.makedirs(dest)
671    print(f"1) Extraction d'une copie propre de HEAD vers {dest}")
672    with tempfile.TemporaryFile() as tmp:
673        subprocess.run(["git", "-C", RACINE, "archive", "HEAD"], stdout=tmp, check=True)
674        tmp.seek(0)
675        with tarfile.open(fileobj=tmp, mode="r") as tar:
676            tar.extractall(dest)  # noqa: S202 — archive produite par git, pas d'attaque Zip Slip ici
677
678    print("2) Initialisation du dépôt et de l'identité")
679    subprocess.run(["git", "init", "-b", "main", dest], capture_output=True, check=True)
680    git(dest, "config", "user.name", nom)
681    git(dest, "config", "user.email", email)
682
683    total = 0
684    print(f"3) Rejeu des {len(PLAN)} commits")
685    for e in PLAN:
686        git(dest, "add", "--", *e["globs"])
687        staged = git(dest, "diff", "--cached", "--name-only").stdout.split("\n")
688        ajoutes = [s for s in staged if s]
689        if not ajoutes:
690            raise SystemExit(f"{e['numero']} : aucun fichier stagé pour {e['globs']}")
691        git(dest, "commit", "-q", "-m", e["titre"], *[part for p in e["corps"] for part in ("-m", p)])
692        total += len(ajoutes)
693        print(f"   {e['numero']:3s} {len(ajoutes):3d} fichier(s) — {e['titre'][:66]}")
694        if "tag" in e:
695            git(dest, "tag", "-a", e["tag"], "-m", TAGS[e["tag"]])
696
697    print("4) Contrôle de couverture (arbre final == arbre d'origine)")
698    attendus = sorted(git(RACINE, "ls-files").stdout.split("\n")[:-1])
699    obtenus = sorted(git(dest, "ls-files").stdout.split("\n")[:-1])
700    if attendus != obtenus:
701        manquants = set(attendus) - set(obtenus)
702        en_trop = set(obtenus) - set(attendus)
703        raise SystemExit(f"COUVERTURE INCORRECTE ! manquants={sorted(manquants)} en_trop={sorted(en_trop)}")
704    restant = git(dest, "status", "--porcelain").stdout.strip()
705    if restant:
706        raise SystemExit(f"Fichiers non suivis restants dans la copie :\n{restant}")
707
708    nb_commits = git(dest, "rev-list", "--count", "HEAD").stdout.strip()
709    print()
710    print(f"✅ {nb_commits} commits, {total} fichiers, {len(attendus)} fichiers suivis identiques à l'origine.")
711    print(f"   Copie prête : {dest}")
712    print()
713    print("Prochaine étape (projet vide créé sur GitLab) :")
714    print(f"  cd {dest}")
715    print("  git remote add origin https://gitlab.example.fr/<groupe>/minishop-sae.git")
716    print("  git push -u origin main --tags")
